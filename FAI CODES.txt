# Means-Ends Analysis for a Domestic Robot

# Initial state
place1 = ["A", "B", "C"]
place2 = []
robot = "Place1"

# Goal: Stack A, B, C at Place2
goal = ["A", "B", "C"]

print("Initial State:")
print("Place1:", place1)
print("Place2:", place2)
print("Robot at:", robot)

print("\nPlanning using Means-Ends Analysis")

# Step 1: Transfer object A
print("\nStep 1: Pick up A")
place1.remove("A")
print("A picked up by robot")

print("Step 2: Move to Place2")
robot = "Place2"

print("Step 3: Put down A")
place2.append("A")
print("Place2:", place2)

# Step 2: Transfer object B
print("\nStep 4: Move to Place1")
robot = "Place1"

print("Step 5: Pick up B")
place1.remove("B")
print("B picked up by robot")

print("Step 6: Move to Place2")
robot = "Place2"

print("Step 7: Stack B on A")
place2.append("B")
print("Place2:", place2)

# Step 3: Transfer object C
print("\nStep 8: Move to Place1")
robot = "Place1"

print("Step 9: Pick up C")
place1.remove("C")
print("C picked up by robot")

print("Step 10: Move to Place2")
robot = "Place2"

print("Step 11: Stack C on B")
place2.append("C")
print("Place2:", place2)

# Final state
print("\nFinal State:")
print("Place1:", place1)
print("Place2:", place2)
print("Robot at:", robot)

if place2 == goal:
    print("Goal Achieved!")
else:
    print("Goal Not Achieved!")

def min_value(a, b, alpha, node):
    best = float('inf')

    for i, value in enumerate([a, b]):
        best = min(best, value)
        beta = best

        if beta <= alpha:
            print("Pruned branch at", node)
            return best, True

    return best, False


print("ALPHA-BETA PRUNING")

print("\nEnter values for 4 leaf nodes:")
a = int(input("A1: "))
b = int(input("A2: "))
c = int(input("B1: "))
d = int(input("B2: "))

alpha = float('-inf')

# First MIN node
left, p1 = min_value(a, b, alpha, "A")

alpha = max(alpha, left)

# Second MIN node
right, p2 = min_value(c, d, alpha, "B")

# Root MAX node
result = max(left, right)

print("\nGAME TREE")
print("             MAX =", result)
print("           /       \\")
print("        MIN         MIN")
print("       /  \\        /  \\")
print("      ", a, "  ", b, "     ", c, "  ", d)

print("\nFinal Answer:", result)
print("Alpha-Beta Pruning completed.")

text = input("Enter an assertive sentence: ")

# Stage 1: Sentence Segmentation
sentences = text.split(".")
print("\n1. Sentence Segmentation:")
print(sentences)

for sentence in sentences:
    if sentence.strip() == "":
        continue

    # Stage 2: Word Tokenization
    words = sentence.split()
    print("\n2. Word Tokenization:")
    print(words)

    # Stage 3: POS Tagging
    print("\n3. POS Tagging:")
    for word in words:
        w = word.lower().strip(",!?")

        if w in ["the", "a", "an"]:
            print(word, "- Determiner")
        elif w in ["is", "am", "are", "was", "were", "reads", "plays", "eats", "likes"]:
            print(word, "- Verb")
        elif w in ["he", "she", "they", "we", "i"]:
            print(word, "- Pronoun")
        elif w in ["good", "beautiful", "big", "small"]:
            print(word, "- Adjective")
        else:
            print(word, "- Noun")

    # Stage 4: Simple Syntactic Parsing
    print("\n4. Syntactic Parsing:")
    print("Sentence")
    print(" |")
    print(" +-- Subject:", words[0])
    if len(words) > 1:
        print(" +-- Predicate:", " ".join(words[1:]))

print("\nSyntax Analysis Completed")

def hill_climbing():
    # Initial state
    current = 0

    # Values of neighboring states
    values = [1, 3, 5, 7, 6, 4, 2]

    print("Hill Climbing Search")

    while True:
        print("Current state:", current,
              "Value:", values[current])

        # Check left neighbor
        left = current - 1
        right = current + 1

        best = current

        if left >= 0 and values[left] > values[best]:
            best = left

        # Check right neighbor
        if right < len(values) and values[right] > values[best]:
            best = right

        # Stop if no better neighbor exists
        if best == current:
            print("Maximum value found:", values[current])
            print("Position:", current)
            break

        current = best


hill_climbing()
